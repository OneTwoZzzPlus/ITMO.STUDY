CREATE SCHEMA IF NOT EXISTS "Countries" AUTHORIZATION postgres;
CREATE SCHEMA IF NOT EXISTS "EmployeesDepartments" AUTHORIZATION postgres;

CREATE TABLE  "Countries"."REGIONS"
   ("REGION_ID" SMALLINT,
    "REGION_NAME" VARCHAR(35),
    CONSTRAINT "REG_ID_PK" PRIMARY KEY ("REGION_ID")
   );

CREATE TABLE  "Countries"."COUNTRIES"
   ("COUNTRY_ID" SMALLINT, 
	"REGION_ID" SMALLINT NOT NULL, 
	"COUNTRY_NAME" VARCHAR(70) CONSTRAINT "CNAME_NN" NOT NULL, 
	"COUNTRY_TRANSLATED_NAME" VARCHAR(40), 
	"LOCATION" VARCHAR(90), 
	"CAPITOL" VARCHAR(50), 
	"AREA" BIGINT, 
	"COASTLINE" INT, 
	"LOWEST_ELEVATION" INT, 
	"LOWEST_ELEV_NAME" VARCHAR(70), 
	"HIGHEST_ELEVATION" INT, 
	"HIGHEST_ELEV_NAME" VARCHAR(50), 
	"DATE_OF_INDEPENDENCE" VARCHAR(30), 
	"NATIONAL_HOLIDAY_NAME" VARCHAR(200), 
	"NATIONAL_HOLIDAY_DATE" VARCHAR(30), 
	"POPULATION" BIGINT, 
	"POPULATION_GROWTH_RATE" VARCHAR(10), 
	"LIFE_EXPECT_AT_BIRTH" DECIMAL(6,2), 
	"MEDIAN_AGE" DECIMAL(6,2), 
	"AIRPORTS" INT, 
	"CLIMATE" VARCHAR(1000), 
	"FIPS_ID" CHAR(2), 
	"INTERNET_EXTENSION" VARCHAR(3), 
	"FLAG" BYTEA, 
	"CURRENCY_CODE" VARCHAR(7) NOT NULL, 
	CONSTRAINT "CRTY_PK" PRIMARY KEY ("COUNTRY_ID")
   );
  
DROP TABLE IF EXISTS "EmployeesDepartments"."LOCATIONS";
CREATE TABLE   "EmployeesDepartments"."LOCATIONS"
   ("LOCATION_ID" SMALLINT,
    "STREET_ADDRESS" VARCHAR(40),
    "POSTAL_CODE" VARCHAR(12),
    "CITY" VARCHAR(30) CONSTRAINT "LOC_CITY_NN" NOT NULL,
    "STATE_PROVINCE" VARCHAR(25),
    "COUNTRY_ID" SMALLINT,
    CONSTRAINT "LOC_ID_PK" PRIMARY KEY ("LOCATION_ID")
   );   

CREATE TABLE  "Countries"."CURRENCIES" 
   ("CURRENCY_CODE" VARCHAR(7), 
	"CURRENCY_NAME" VARCHAR(40) CONSTRAINT "CURR_NN" NOT NULL, 
	"COMMENTS" VARCHAR(150), 
	CONSTRAINT "CURR_PK" PRIMARY KEY ("CURRENCY_CODE")
   );
   
CREATE TABLE  "Countries"."LANGUAGES"
   ("LANGUAGE_ID" SMALLINT, 
	"LANGUAGE_NAME" VARCHAR(50) CONSTRAINT "LANG_LANGNAME_NN" NOT NULL, 
	CONSTRAINT "LANG_PK" PRIMARY KEY ("LANGUAGE_ID")
   );
   
CREATE TABLE  "Countries"."SPOKEN_LANGUAGES"
   ("COUNTRY_ID" SMALLINT, 
	"LANGUAGE_ID" SMALLINT, 
	"OFFICIAL" VARCHAR(2), 
	"COMMENTS" VARCHAR(512), 
	CONSTRAINT "SPOKEN_LANG_PK" PRIMARY KEY ("COUNTRY_ID", "LANGUAGE_ID")
   );

DROP TABLE IF EXISTS "EmployeesDepartments"."DEPARTMENTS";
CREATE TABLE  "EmployeesDepartments"."DEPARTMENTS"
   ("DEPARTMENT_ID" SMALLINT GENERATED ALWAYS AS IDENTITY ( INCREMENT 1 START 1),

    "DEPARTMENT_NAME" VARCHAR(30) CONSTRAINT "DEPT_NAME_NN" NOT NULL,
    "MANAGER_ID" INT,
    "LOCATION_ID" SMALLINT,
     CONSTRAINT "DEPT_ID_PK" PRIMARY KEY ("DEPARTMENT_ID")
   );

CREATE TABLE  "EmployeesDepartments"."JOBS"
   ("JOB_ID" VARCHAR(10),
    "JOB_TITLE" VARCHAR(35) CONSTRAINT "JOB_TITLE_NN" NOT NULL,
    "MIN_SALARY" INT,
    "MAX_SALARY" INT,
     CONSTRAINT "JOB_ID_PK" PRIMARY KEY ("JOB_ID")
   );

DROP TABLE IF EXISTS "EmployeesDepartments"."EMPLOYEES";	
CREATE TABLE "EmployeesDepartments"."EMPLOYEES"
   ("EMPLOYEE_ID" INT NOT NULL GENERATED ALWAYS AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 CACHE 1 ),
    "FIRST_NAME" VARCHAR(20),
    "LAST_NAME" VARCHAR(25) CONSTRAINT "EMP_LAST_NAME_NN" NOT NULL,
    "EMAIL" VARCHAR(25) CONSTRAINT "EMP_EMAIL_NN" NOT NULL,
    "PHONE_NUMBER" VARCHAR(20),
    "HIRE_DATE" date NOT NULL DEFAULT CURRENT_DATE,
    "JOB_ID" VARCHAR(10) CONSTRAINT "EMP_JOB_NN" NOT NULL,
    "SALARY" DECIMAL(8,2),
    "COMMISSION_PCT" DECIMAL(2,2),
    "MANAGER_ID" INT,
    "DEPARTMENT_ID" SMALLINT,
	"BONUS" VARCHAR(5),
	CONSTRAINT "EMP_ID_PK" PRIMARY KEY ("EMPLOYEE_ID")
  );

CREATE TABLE  "EmployeesDepartments"."JOB_GRADES"
   ("GRADE_LEVEL" VARCHAR(3),
    "LOWEST_SAL" DECIMAL,
    "HIGHEST_SAL" DECIMAL
   );

CREATE TABLE  "EmployeesDepartments"."JOB_HISTORY"
   ("EMPLOYEE_ID" INT CONSTRAINT "JHIST_EMPLOYEE_NN" NOT NULL,
    "START_DATE" TIMESTAMP(0) CONSTRAINT "JHIST_START_DATE_NN" NOT NULL,
    "END_DATE" TIMESTAMP(0) CONSTRAINT "JHIST_END_DATE_NN" NOT NULL,
    "JOB_ID" VARCHAR(10) CONSTRAINT "JHIST_JOB_NN" NOT NULL,
    "DEPARTMENT_ID" SMALLINT,
    CONSTRAINT "JHIST_EMP_ID_ST_DATE_PK" PRIMARY KEY ("EMPLOYEE_ID", "START_DATE")
  );
